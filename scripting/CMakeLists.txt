cmake_minimum_required(VERSION 3.24)
project(zEngineScripting LANGUAGES CXX)
add_library(zEngineScripting STATIC src/Script.cpp)
target_include_directories(zEngineScripting PUBLIC include)
target_compile_features(zEngineScripting PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(zEngineScripting PRIVATE /W4 /permissive- /EHsc)
else()
    target_compile_options(zEngineScripting PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(zEngineScriptTests tests/ScriptTests.cpp)
    target_link_libraries(zEngineScriptTests PRIVATE zEngineScripting)
    if(MSVC)
        target_compile_options(zEngineScriptTests PRIVATE /utf-8)
    endif()
    target_compile_definitions(zEngineScriptTests PRIVATE SCRIPT_EXAMPLES="${CMAKE_CURRENT_SOURCE_DIR}/examples" ZSCRIPT_TYPES_JSON="${CMAKE_CURRENT_SOURCE_DIR}/types.json")
    add_test(NAME ScriptRuntime COMMAND zEngineScriptTests)
    set_tests_properties(ScriptRuntime PROPERTIES TIMEOUT 20)
endif()
